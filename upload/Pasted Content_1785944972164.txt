Both built. The loyalty handlers close the points loop (JOIN earns a bonus, visits earn points, BALANCE shows them, REDEEM spends them), and the Super Admin layer gives you tenant oversight, CSV prospect intake, platform broadcasts, and a webhook debug log.

# Feature 1 · Loyalty handlers + earn-on-visit

## 1 · Loyalty service (JOIN / BALANCE / STOP / earn-on-visit)

```ts
// src/modules/loyalty/service.ts
import { db } from '@/lib/db';
import { guests, tenants, loyaltyTransactions, rewardsCatalog, reservations } from '@/lib/db/schema';
import { eq, and, asc } from 'drizzle-orm';
import { sendMessageToGuest } from '@/modules/whatsapp/send';

export const JOIN_BONUS_POINTS = 10;
export const EARN_CENTS_PER_POINT = 1000; // 1 point per R10 spent
export const FLAT_VISIT_POINTS = 5;       // fallback when no spend is captured

async function findGuestByPhone(tenantId: string, phone: string) {
  if (!db) return null;
  const [g] = await db.select().from(guests)
    .where(and(eq(guests.tenantId, tenantId), eq(guests.phone, phone))).limit(1);
  return g ?? null;
}

async function getCurrency(tenantId: string): Promise<string> {
  if (!db) return 'Points';
  const [t] = await db.select({ currencyName: tenants.currencyName }).from(tenants).where(eq(tenants.id, tenantId)).limit(1);
  return t?.currencyName ?? 'Points';
}

/* ── JOIN ───────────────────────────────────────────────────────────── */
export async function handleJoin(tenantId: string, phone: string) {
  if (!db) throw new Error('DATABASE_NOT_CONFIGURED');
  const currency = await getCurrency(tenantId);
  const [tenant] = await db.select().from(tenants).where(eq(tenants.id, tenantId)).limit(1);
  const existing = await findGuestByPhone(tenantId, phone);

  if (existing) {
    if (existing.status === 'opted_out') {
      await db.update(guests).set({ status: 'active', optedOutAt: null }).where(eq(guests.id, existing.id));
      await sendMessageToGuest(tenantId, phone,
        `Welcome back! 🎉 You're re-joined with ${existing.pointsBalance} ${currency} still saved. Text BALANCE anytime.`,
        { guestId: existing.id, handledBy: 'keyword' });
      return { status: 'reactivated' };
    }
    await sendMessageToGuest(tenantId, phone,
      `You're already in! You have ${existing.pointsBalance} ${currency}. Text BALANCE to check, REDEEM to claim a reward.`,
      { guestId: existing.id, handledBy: 'keyword' });
    return { status: 'already_member' };
  }

  const [guest] = await db.insert(guests).values({
    tenantId, phone, status: 'active',
    pointsBalance: JOIN_BONUS_POINTS,
    totalVisits: 0, totalSpent: 0,
    joinedAt: new Date(), source: 'whatsapp',
  }).returning();

  await db.insert(loyaltyTransactions).values({
    tenantId, guestId: guest.id, type: 'join_bonus',
    points: JOIN_BONUS_POINTS, description: 'Welcome bonus for joining',
  });

  await sendMessageToGuest(tenantId, phone,
    `Welcome to ${tenant?.name ?? 'our rewards'}! 🎉 You've joined and earned ${JOIN_BONUS_POINTS} ${currency} to start. Text BALANCE to check your points, or REDEEM to claim a reward.`,
    { guestId: guest.id, handledBy: 'keyword' });
  return { status: 'joined', guestId: guest.id };
}

/* ── BALANCE ────────────────────────────────────────────────────────── */
export async function handleBalance(tenantId: string, phone: string) {
  if (!db) throw new Error('DATABASE_NOT_CONFIGURED');
  const currency = await getCurrency(tenantId);
  const guest = await findGuestByPhone(tenantId, phone);

  if (!guest) {
    await sendMessageToGuest(tenantId, phone,
      `You're not in yet! Text JOIN to start earning ${currency}. 😊`,
      { handledBy: 'keyword' });
    return { status: 'not_member' };
  }

  const rewards = await db.select().from(rewardsCatalog)
    .where(and(eq(rewardsCatalog.tenantId, tenantId), eq(rewardsCatalog.isActive, true)))
    .orderBy(asc(rewardsCatalog.pointsCost));

  const affordable = rewards.filter((r) => r.pointsCost <= guest.pointsBalance);
  const next = rewards.find((r) => r.pointsCost > guest.pointsBalance) ?? null;

  let msg = `You have ${guest.pointsBalance} ${currency}. ⭐`;
  if (affordable.length) {
    msg += `\n\nYou can REDEEM: ${affordable.map((r) => r.name).join(', ')}.`;
  } else if (next) {
    msg += `\n\n${next.pointsCost - guest.pointsBalance} more to unlock "${next.name}". Keep visiting!`;
  }

  await sendMessageToGuest(tenantId, phone, msg, { guestId: guest.id, handledBy: 'keyword' });
  return { status: 'sent' };
}

/* ── STOP ───────────────────────────────────────────────────────────── */
export async function handleStop(tenantId: string, phone: string) {
  if (!db) throw new Error('DATABASE_NOT_CONFIGURED');
  const guest = await findGuestByPhone(tenantId, phone);

  if (!guest) {
    await sendMessageToGuest(tenantId, phone, `You're not subscribed, so you're all set. 👍`, { handledBy: 'keyword' });
    return { status: 'not_member' };
  }

  await db.update(guests).set({ status: 'opted_out', optedOutAt: new Date() }).where(eq(guests.id, guest.id));
  await sendMessageToGuest(tenantId, phone,
    `You're opted out. We'll miss you! Your points are saved if you ever come back. Text JOIN anytime to rejoin. 💛`,
    { guestId: guest.id, handledBy: 'keyword' });
  return { status: 'opted_out' };
}

/* ── EARN ON VISIT ──────────────────────────────────────────────────── */
export async function earnPointsForVisit(tenantId: string, guestId: string, amountSpentCents?: number) {
  if (!db) throw new Error('DATABASE_NOT_CONFIGURED');
  const [guest] = await db.select().from(guests)
    .where(and(eq(guests.id, guestId), eq(guests.tenantId, tenantId))).limit(1);
  if (!guest) return { earned: 0 };

  let earned: number;
  let description: string;
  if (amountSpentCents && amountSpentCents > 0) {
    earned = Math.max(1, Math.floor(amountSpentCents / EARN_CENTS_PER_POINT));
    description = `Earned from visit (R${Math.round(amountSpentCents / 100)} spent)`;
  } else {
    earned = FLAT_VISIT_POINTS;
    description = 'Visit bonus';
  }

  await db.insert(loyaltyTransactions).values({ tenantId, guestId, type: 'earn', points: earned, description });
  const newBalance = guest.pointsBalance + earned;
  await db.update(guests).set({
    pointsBalance: newBalance,
    totalVisits: (guest.totalVisits ?? 0) + 1,
    lastVisitAt: new Date(),
    totalSpent: (guest.totalSpent ?? 0) + (amountSpentCents ?? 0),
  }).where(eq(guests.id, guestId));

  const currency = await getCurrency(tenantId);
  try {
    await sendMessageToGuest(tenantId, guest.phone,
      `Thanks for visiting! 🎉 You just earned ${earned} ${currency}. New balance: ${newBalance} ${currency}. Text REDEEM to claim a reward.`,
      { guestId, handledBy: 'automation' });
  } catch (e) {
    console.warn('[loyalty] earn notify failed', e);
  }
  return { earned, newBalance };
}

/* ── COMPLETE A VISIT (marks reservation completed + awards points) ─── */
export async function completeVisitAndEarn(tenantId: string, reservationId: string, amountSpentCents?: number) {
  if (!db) throw new Error('DATABASE_NOT_CONFIGURED');
  const [reservation] = await db.select().from(reservations)
    .where(and(eq(reservations.id, reservationId), eq(reservations.tenantId, tenantId))).limit(1);
  if (!reservation) throw new Error('RESERVATION_NOT_FOUND');
  if (reservation.status === 'completed') return { alreadyCompleted: true };

  await db.update(reservations).set({ status: 'completed', completedAt: new Date() })
    .where(eq(reservations.id, reservationId));

  if (!reservation.guestId) return { completed: true, earned: 0 };
  const earnResult = await earnPointsForVisit(tenantId, reservation.guestId, amountSpentCents);
  return { completed: true, ...earnResult };
}
```

```ts
// src/app/api/v1/admin/prospects/upload/route.ts  (CSV)
import { NextResponse } from 'next/server';
import crypto from 'crypto';
import { db } from '@/lib/db';
import { prospects } from '@/lib/db/schema';
import { eq } from 'drizzle-orm';
import { requireSuperAdmin } from '@/shared/utils/super-admin';

function normalizePhone(raw: string): string {
  let d = raw.replace(/\D/g, '');
  if (d.startsWith('0')) d = '27' + d.slice(1);
  return d;
}
function splitCsvLine(line: string): string[] {
  const out: string[] = []; let cur = ''; let inQ = false;
  for (let i = 0; i < line.length; i++) {
    const c = line[i];
    if (c === '"') { if (inQ && line[i + 1] === '"') { cur += '"'; i++; } else inQ = !inQ; }
    else if (c === ',' && !inQ) { out.push(cur); cur = ''; }
    else cur += c;
  }
  out.push(cur);
  return out;
}
function parseCsv(text: string) {
  const lines = text.split(/\r?\n/).filter((l) => l.trim().length > 0);
  if (!lines.length) return [];
  const header = splitCsvLine(lines[0]).map((h) => h.trim().toLowerCase());
  const nameIdx = header.findIndex((h) => ['business_name','business name','businessname','name','restaurant'].includes(h));
  const phoneIdx = header.findIndex((h) => ['phone','phone number','phonenumber','mobile','number'].includes(h));
  const emailIdx = header.findIndex((h) => ['email','e-mail'].includes(h));
  const industryIdx = header.findIndex((h) => ['industry','type','category'].includes(h));
  const rows = [];
  for (let i = 1; i < lines.length; i++) {
    const cols = splitCsvLine(lines[i]);
    rows.push({
      businessName: nameIdx >= 0 ? (cols[nameIdx] ?? '').trim() : '',
      phone: phoneIdx >= 0 ? (cols[phoneIdx] ?? '').trim() : '',
      email: emailIdx >= 0 ? (cols[emailIdx] ?? '').trim() : '',
      industry: industryIdx >= 0 ? (cols[industryIdx] ?? '').trim() : '',
    });
  }
  return rows;
}

export async function POST(request: Request) {
  try {
    await requireSuperAdmin();
    if (!db) return NextResponse.json({ error: { code: 'DATABASE_NOT_CONFIGURED' } }, { status: 503 });

    const file = (await request.formData()).get('file') as File | null;
    if (!file) return NextResponse.json({ error: { code: 'NO_FILE' } }, { status: 400 });

    const rows = parseCsv(await file.text());
    let created = 0, skipped = 0;
    for (const row of rows) {
      if (!row.businessName || !row.phone) { skipped++; continue; }
      const phone = normalizePhone(row.phone);
      const [existing] = await db.select({ id: prospects.id }).from(prospects).where(eq(prospects.phone, phone)).limit(1);
      if (existing) { skipped++; continue; }
      await db.insert(prospects).values({
        businessName: row.businessName, phone,
        email: row.email || null, industry: row.industry || null,
        status: 'new', claimToken: crypto.randomUUID(),
      });
      created++;
    }
    return NextResponse.json({ created, skipped });
  } catch (e) {
    const msg = e instanceof Error ? e.message : 'INTERNAL';
    const status = msg === 'UNAUTHORIZED' ? 401 : msg === 'FORBIDDEN' ? 403 : 500;
    return NextResponse.json({ error: { code: msg } }, { status });
  }
}
```

```ts
    await requireSuperAdmin();
    if (!db) return NextResponse.json({ error: { code: 'DATABASE_NOT_CONFIGURED' } }, { status: 503 });

        if (res.success) sent++; else failed++;
      } catch { failed++; }
    }
    return NextResponse.json({ sent, failed, total: active.length });
  } catch (e) {
    const msg = e instanceof Error ? e.message : 'INTERNAL';
    const status = msg === 'UNAUTHORIZED' ? 401 : msg === 'FORBIDDEN' ? 403 : 500;
```

## Honest notes & follow-ups

- **Broadcast sends from each tenant's own WhatsApp number** to their owner (reuses the gateway). A dedicated platform sender number is a cleaner follow-up.
- **Broadcast is sequential** — fine for a handful of tenants, move to a queue at scale.
- **CSV parser is built-in** (handles quoted fields). Swap to papaparse if you need more robust parsing.
- **CSV upload isn't rate-limited** — it's super-admin-only, but consider a guard.
- **REDEEM while opted-out:** decide whether STOPped guests can still redeem. Currently `processRedeem` doesn't block it — add a guard if you want.
