```ts
const schema = z.object({
  capacity: z.number().int().min(0).max(1000).optional(),
  avgSpend: z.number().int().min(0).optional(), // cents
  openingHours: z.string().max(500).optional(),
});

export async function GET() {
  try {
    const { tenant } = await requireTenantContext();
    return NextResponse.json({
      capacity: tenant.capacity, avgSpend: tenant.avgSpend,
      openingHours: (tenant.openingHours as any)?.text ?? '',
    });
  } catch (e) {
    const msg = e instanceof Error ? e.message : 'INTERNAL';
    return NextResponse.json({ error: { code: msg } }, { status: msg === 'UNAUTHORIZED' ? 401 : msg === 'NO_TENANT' ? 403 : 500 });
  }

export async function PATCH(request: Request) {
  try {
    const { tenant } = await requireTenantContext();
    if (!db) return NextResponse.json({ error: { code: 'DATABASE_NOT_CONFIGURED' } }, { status: 503 });
    const parsed = schema.safeParse(await request.json().catch(() => null));
    if (!parsed.success) return NextResponse.json({ error: { code: 'VALIDATION' } }, { status: 400 });

    if (parsed.data.capacity !== undefined) updates.capacity = parsed.data.capacity;
    if (parsed.data.avgSpend !== undefined) updates.avgSpend = parsed.data.avgSpend;
    if (parsed.data.openingHours !== undefined) updates.openingHours = { text: parsed.data.openingHours };
    await db.update(tenants).set(updates).where(eq(tenants.id, tenant.id));
    return NextResponse.json({ success: true });
  } catch (e) {
    const msg = e instanceof Error ? e.message : 'INTERNAL';
```

```ts
// src/lib/integrations/payfast/signature.ts
import { createHash } from 'crypto';

// Order-preserved MD5 (Custom Integration). NOT the REST API's alphabetical scheme.
export function generateSignature(orderedFields: [string, string][], passphrase: string): string {
  const parts = orderedFields
    .filter(([k, v]) => k !== 'signature' && v !== undefined && v !== null && v !== '')
    .map(([k, v]) => `${k}=${encodeURIComponent(v)}`);
  const str = parts.join('&') + (passphrase ? `&passphrase=${encodeURIComponent(passphrase)}` : '');
  return createHash('md5').update(str).digest('hex');
}
```

```ts
// src/app/api/v1/billing/checkout/route.ts
import { NextResponse } from 'next/server';
import { requireTenantContext } from '@/shared/utils/tenant-context';
import { PLANS, PlanKey } from '@/lib/integrations/payfast/plans';
import { generateSignature } from '@/lib/integrations/payfast/signature';

export async function POST(request: Request) {
  try {
    const { tenant } = await requireTenantContext();
    const body = await request.json().catch(() => ({}));
    const plan = PLANS[body.plan as PlanKey];
    if (!plan) return NextResponse.json({ error: { code: 'INVALID_PLAN' } }, { status: 400 });

    const merchantId = process.env.PAYFAST_MERCHANT_ID;
    const merchantKey = process.env.PAYFAST_MERCHANT_KEY;
    const passphrase = process.env.PAYFAST_PASSPHRASE || '';
    const appUrl = process.env.NEXT_PUBLIC_APP_URL;
    if (!merchantId || !merchantKey || !appUrl) {
      return NextResponse.json({ error: { code: 'PAYFAST_NOT_CONFIGURED' } }, { status: 503 });
    }
    const isSandbox = (process.env.PAYFAST_MODE || 'sandbox') === 'sandbox';
    const processUrl = isSandbox ? 'https://sandbox.payfast.co.za/eng/process' : 'https://www.payfast.co.za/eng/process';

    const fields: Record<string, string> = {
      merchant_id: merchantId,
      merchant_key: merchantKey,
      return_url: `${appUrl}/billing?status=return`,
      cancel_url: `${appUrl}/billing?status=cancelled`,
      notify_url: `${appUrl}/api/webhooks/payfast`,
      m_payment_id: `orderly_${tenant.id}`,
      amount: (plan.amountCents / 100).toFixed(2),
      item_name: `Orderly ${plan.name} (monthly)`,
      subscription_type: '1',   // recurring
      frequency: '3',           // monthly
      cycles: '0',              // until cancelled
    };
    const order = Object.keys(fields);
    const signature = generateSignature(order.map((k) => [k, fields[k]]), passphrase);
    return NextResponse.json({ url: processUrl, fields: { ...fields, signature } });
  } catch (e) {
    const msg = e instanceof Error ? e.message : 'INTERNAL';
    return NextResponse.json({ error: { code: msg } }, { status: msg === 'UNAUTHORIZED' ? 401 : msg === 'NO_TENANT' ? 403 : 500 });
  }
}
```

```ts
import { generateSignature } from '@/lib/integrations/payfast/signature';
import { logWebhookEvent } from '@/lib/webhooks/log';

const PAYFAST_IPS = ['197.97.145.144', '41.74.179.192', '102.130.112.']; // prefix match; keep updated

export async function POST(request: Request) {
  const raw = await request.text();
  const pairs: [string, string][] = [...new URLSearchParams(raw).entries()];
  const get = (k: string) => pairs.find(([key]) => key === k)?.[1] ?? '';
  await logWebhookEvent({ source: 'payfast', eventType: get('payment_status'), rawPayload: Object.fromEntries(pairs), processed: false });

  const passphrase = process.env.PAYFAST_PASSPHRASE || '';
  const receivedSig = get('signature');
  const expectedSig = generateSignature(pairs.filter(([k]) => k !== 'signature'), passphrase);

  // 1) Signature
  if (receivedSig !== expectedSig) return new NextResponse('INVALID_SIGNATURE', { status: 400 });

  // 2) Source IP (soft by default; set PAYFAST_ENFORCE_IP=true to hard-enforce)
  const ip = request.headers.get('x-forwarded-for')?.split(',')[0]?.trim() || '';
  const ipOk = PAYFAST_IPS.some((p) => ip.startsWith(p));
  if (!ipOk && process.env.PAYFAST_ENFORCE_IP === 'true') return new NextResponse('INVALID_IP', { status: 403 });

  // 3) Amount matches a known plan
  const amountGross = parseFloat(get('amount_gross'));
  const matched = Object.values(PLANS).find((p) => Math.abs(p.amountCents / 100 - amountGross) < 0.01);
  if (!matched) return new NextResponse('AMOUNT_MISMATCH', { status: 400 });

  // 4) Server confirmation with PayFast
  const isSandbox = (process.env.PAYFAST_MODE || 'sandbox') === 'sandbox';
  const validateUrl = isSandbox ? 'https://sandbox.payfast.co.za/eng/query/validate' : 'https://www.payfast.co.za/eng/query/validate';
  try {
    const res = await fetch(validateUrl, { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: raw });
    const valid = (await res.text()).trim() === 'VALID';
    if (!valid) return new NextResponse('NOT_VALID', { status: 400 });
  } catch { return new NextResponse('VALIDATE_FAILED', { status: 502 }); }

  // Activate the tenant's subscription
  if (get('payment_status') === 'COMPLETE' && db) {
    const mPaymentId = get('m_payment_id'); // orderly_{tenantId}
    const tenantId = mPaymentId.replace('orderly_', '');
    await db.update(tenants).set({
      plan: matched.key, status: 'active',
      payfastToken: get('token') || null,
      trialEndsAt: null,
    } as any).where(eq(tenants.id, tenantId));
  }
  return new NextResponse('OK', { status: 200 });
}
```

```tsx
// src/components/settings/billing-client.tsx
'use client';
import { useState } from 'react';
import { Button } from '@/components/ui/button';
import { Loader2, Check } from 'lucide-react';

const PLANS = [
  { key: 'starter', name: 'Starter', price: 'R299/mo', perks: ['Loyalty + rewards', 'WhatsApp concierge', 'Bookings', 'Campaigns'] },
  { key: 'growth', name: 'Growth', price: 'R499/mo', perks: ['Everything in Starter', 'Advanced campaigns', 'Daily insights', 'Priority support'] },
];

export function BillingClient() {
  const [loading, setLoading] = useState<string | null>(null);

  async function checkout(plan: string) {
    setLoading(plan);
    try {
      const res = await fetch('/api/v1/billing/checkout', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ plan }) });
      const { url, fields } = await res.json();
      // auto-submit a form to PayFast
      const form = document.createElement('form');
      form.method = 'POST'; form.action = url;
      Object.entries(fields).forEach(([k, v]) => {
        const input = document.createElement('input');
        input.type = 'hidden'; input.name = k; input.value = String(v);
        form.appendChild(input);
      });
      document.body.appendChild(form); form.submit();
    } catch { setLoading(null); }
  }

  return (
    <div className="min-h-screen bg-[#16110d] text-[#f5efe6]">
      <div className="mx-auto max-w-3xl px-6 py-10">
        <h1 className="text-2xl font-semibold [font-family:var(--font-display)]">Choose your plan</h1>
        <p className="mt-1 text-sm text-[#a89b8a]">Your 14-day trial keeps working until you subscribe.</p>
        <div className="mt-6 grid gap-4 md:grid-cols-2">
          {PLANS.map((p) => (
            <div key={p.key} className="rounded-2xl border border-white/5 bg-[#211a13] p-6">
              <h2 className="text-lg font-semibold">{p.name}</h2>
              <p className="mt-1 text-2xl font-bold text-[#e8722a]">{p.price}</p>
              <ul className="mt-4 space-y-2 text-sm text-[#a89b8a]">
                {p.perks.map((x) => <li key={x} className="flex items-center gap-2"><Check className="h-4 w-4 text-emerald-400" />{x}</li>)}
              </ul>
              <Button onClick={() => checkout(p.key)} disabled={loading === p.key} className="mt-5 w-full bg-[#e8722a] hover:bg-[#f0823a]">
                {loading === p.key && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}Subscribe
              </Button>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}
```
