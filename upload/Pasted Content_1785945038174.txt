```ts
export async function sendPlatformMessage(toPhone: string, content: string): Promise<SendResult> {
  const instanceName = process.env.PLATFORM_INSTANCE_NAME;
  const instanceToken = process.env.PLATFORM_INSTANCE_TOKEN;
  if (!instanceName || !instanceToken) return { success: false, error: 'PLATFORM_WHATSAPP_NOT_CONFIGURED' };

    return { success: false, error: result.error };
  }
  return { success: true, messageId: result.data?.key?.id };
```

## 2 · Invite endpoint (Super Admin → prospect)

```ts
// src/app/api/v1/admin/prospects/[id]/invite/route.ts
import { NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { prospects } from '@/lib/db/schema';
import { eq } from 'drizzle-orm';
import { requireSuperAdmin } from '@/shared/utils/super-admin';
import { sendPlatformMessage } from '@/modules/whatsapp/send';

export async function POST(_req: Request, { params }: { params: Promise<{ id: string }> }) {
  try {
    await requireSuperAdmin();
    if (!db) return NextResponse.json({ error: { code: 'DATABASE_NOT_CONFIGURED' } }, { status: 503 });
    const { id } = await params;

    const [prospect] = await db.select().from(prospects).where(eq(prospects.id, id)).limit(1);
    if (!prospect) return NextResponse.json({ error: { code: 'NOT_FOUND' } }, { status: 404 });
    if (!prospect.claimToken) return NextResponse.json({ error: { code: 'NO_CLAIM_TOKEN' } }, { status: 400 });

    const appUrl = process.env.NEXT_PUBLIC_APP_URL || '';
    const claimLink = `${appUrl}/claim/${prospect.claimToken}`;
    const message =
      `Hi ${prospect.businessName}! 👋 You've been invited to join Orderly — the system that fills your empty chairs and turns walk-ins into regulars.\n\n` +
      `Claim your restaurant's profile here: ${claimLink}\n\nIt takes about 2 minutes. Your customers download nothing — everything runs on WhatsApp.`;

    const res = await sendPlatformMessage(prospect.phone, message);
    if (!res.success) return NextResponse.json({ error: { code: 'SEND_FAILED', detail: res.error } }, { status: 502 });

    await db.update(prospects).set({ status: 'invited', invitedAt: new Date() }).where(eq(prospects.id, id));
    return NextResponse.json({ success: true });
  } catch (e) {
    const msg = e instanceof Error ? e.message : 'INTERNAL';
    const status = msg === 'UNAUTHORIZED' ? 401 : msg === 'FORBIDDEN' ? 403 : 500;
    return NextResponse.json({ error: { code: msg } }, { status });
  }
}
```

## 4 · Claim server action (prospect → tenant)

```ts
// src/app/claim/[token]/actions.ts
'use server';
import { auth } from '@clerk/nextjs/server';
import { redirect } from 'next/navigation';
import { db } from '@/lib/db';
import { prospects, tenants, profiles } from '@/lib/db/schema';
import { eq } from 'drizzle-orm';

function slugify(s: string): string {
  return s.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '').slice(0, 40) || 'restaurant';
}

export async function claimProspectAction(_prev: { error: string }, formData: FormData) {
  const token = String(formData.get('token') || '');
  const businessName = String(formData.get('businessName') || '').trim();
  const ownerName = String(formData.get('ownerName') || '').trim();
  const phone = String(formData.get('phone') || '').trim();

  const { userId } = await auth();
  if (!userId) return { error: 'Please sign in first.' };
  if (!db) return { error: 'Service is starting up — try again in a moment.' };
  if (!businessName || !ownerName) return { error: 'Please fill in your business and your name.' };

  const [prospect] = await db.select().from(prospects).where(eq(prospects.claimToken, token)).limit(1);
  if (!prospect) return { error: 'This invite link is not valid.' };
  if (prospect.status === 'claimed') return { error: 'This business has already been claimed.' };

  const [existingProfile] = await db.select().from(profiles).where(eq(profiles.clerkId, userId)).limit(1);
  if (existingProfile) return { error: 'You already have a restaurant on Orderly.' };

  const [tenant] = await db.insert(tenants).values({
    name: businessName,
    slug: slugify(businessName),
    industry: prospect.industry ?? 'restaurant',
    phone: phone || prospect.phone,
    status: 'trial',
    plan: 'starter',
    currencyName: 'Points',
    trialEndsAt: new Date(Date.now() + 14 * 24 * 3600 * 1000),
  }).returning();

  await db.insert(profiles).values({
    clerkId: userId, tenantId: tenant.id, role: 'owner',
    fullName: ownerName, phone: phone || null,
  });

  await db.update(prospects).set({ status: 'claimed', tenantId: tenant.id }).where(eq(prospects.id, prospect.id));

  redirect('/dashboard');
}
```

## 7 · Campaigns service (audience + ROI + send)

```ts
// src/modules/campaigns/service.ts
import { db } from '@/lib/db';
import { campaigns, guests, tenants } from '@/lib/db/schema';
import { eq, and, inArray } from 'drizzle-orm';
import { sendMessageToGuest } from '@/modules/whatsapp/send';

export type CampaignType = 'fill_quiet_hours' | 'bring_back_lost' | 'reward_vips';

export const CAMPAIGN_PRESETS: Record<CampaignType, {
  label: string; goal: string; statuses: string[]; defaultMessage: string; conversionRate: number;
}> = {
  fill_quiet_hours: {
    label: 'Fill Quiet Hours',
    goal: 'Fill empty seats during slow periods',
    statuses: ['active'],
    defaultMessage: "It's a little quiet at {restaurant} right now! Come in today, show this message, and get 15% off your meal. See you soon! 🍽️",
    conversionRate: 0.15,
  },
  bring_back_lost: {
    label: 'Bring Back Lost Faces',
    goal: 'Win back guests who have gone quiet',
    statuses: ['at_risk', 'dormant'],
    defaultMessage: "We miss you at {restaurant}! 💛 Come back this week and we'll add 10 bonus points to your account. Text BALANCE to check your points.",
    conversionRate: 0.10,
  },
  reward_vips: {
    label: 'Reward VIPs',
    goal: 'Thank your most loyal guests',
    statuses: ['vip'],
    defaultMessage: "You're one of our VIPs at {restaurant}! ⭐ As a thank-you, enjoy a free reward on us. Text REDEEM to claim it.",
    conversionRate: 0.30,
  },
};

export async function buildAudience(tenantId: string, type: CampaignType) {
  if (!db) throw new Error('DATABASE_NOT_CONFIGURED');
  const preset = CAMPAIGN_PRESETS[type];
  const rows = await db.select().from(guests)
    .where(and(eq(guests.tenantId, tenantId), inArray(guests.status, preset.statuses)));
  return rows.filter((g) => g.status !== 'opted_out');
}

export async function estimateRoi(tenantId: string, type: CampaignType, audienceCount: number) {
  if (!db) throw new Error('DATABASE_NOT_CONFIGURED');
  const [tenant] = await db.select().from(tenants).where(eq(tenants.id, tenantId)).limit(1);
  const avgSpend = tenant?.avgSpend ?? 25000; // default R250/head
  const rate = CAMPAIGN_PRESETS[type].conversionRate;
  const estimatedRevenueCents = Math.round(audienceCount * rate * avgSpend);
  return {
    audienceCount, conversionRate: rate, avgSpendCents: avgSpend,
    estimatedRevenueCents, emptyChairBaselineCents: 0,
    summary: `If ~${Math.round(rate * 100)}% of these ${audienceCount} guests come in, that's about R${Math.round(estimatedRevenueCents / 100)} you'd otherwise miss. Empty chairs earn R0.`,
  };
}

export async function createCampaign(tenantId: string, input: { type: CampaignType; messageTemplate: string }) {
  if (!db) throw new Error('DATABASE_NOT_CONFIGURED');
  const preset = CAMPAIGN_PRESETS[input.type];
  const audience = await buildAudience(tenantId, input.type);
  const roi = await estimateRoi(tenantId, input.type, audience.length);
  const [campaign] = await db.insert(campaigns).values({
    tenantId, name: preset.label, type: input.type, goal: preset.goal,
    messageTemplate: input.messageTemplate,
    audienceFilter: { statuses: preset.statuses },
    audienceCount: audience.length, status: 'draft',
  }).returning();
  return { campaign, roi };
}

export async function sendCampaign(tenantId: string, campaignId: string) {
  if (!db) throw new Error('DATABASE_NOT_CONFIGURED');
  const [campaign] = await db.select().from(campaigns)
    .where(and(eq(campaigns.id, campaignId), eq(campaigns.tenantId, tenantId))).limit(1);
  if (!campaign) throw new Error('CAMPAIGN_NOT_FOUND');
  if (campaign.status === 'sent') throw new Error('ALREADY_SENT');

  const [tenant] = await db.select().from(tenants).where(eq(tenants.id, tenantId)).limit(1);
  const audience = await buildAudience(tenantId, campaign.type as CampaignType);

  await db.update(campaigns).set({ status: 'sending', audienceCount: audience.length }).where(eq(campaigns.id, campaignId));

  let sent = 0, failed = 0;
  for (const guest of audience) {
    const msg = personalize(campaign.messageTemplate, guest, tenant);
    try {
      await sendMessageToGuest(tenantId, guest.phone, msg, { guestId: guest.id, campaignId, handledBy: 'campaign' });
      sent++;
    } catch (e) {
      failed++;
      console.error('[campaign] send failed to', guest.phone, e);
    }
  }

  await db.update(campaigns).set({ status: 'sent', sentAt: new Date() }).where(eq(campaigns.id, campaignId));
  return { sent, failed, total: audience.length };
}

function personalize(template: string, guest: any, tenant: any): string {
  return template
    .replace(/\{name\}/g, guest.name ?? 'friend')
    .replace(/\{points\}/g, String(guest.pointsBalance ?? 0))
    .replace(/\{restaurant\}/g, tenant?.name ?? 'our restaurant');
}

export async function listCampaigns(tenantId: string) {
  if (!db) throw new Error('DATABASE_NOT_CONFIGURED');
  return db.select().from(campaigns)
    .where(eq(campaigns.tenantId, tenantId))
    .orderBy(campaigns.createdAt);
}
```

```ts
export async function sendCampaign(tenantId: string, campaignId: string) {
  if (!db) throw new Error('DATABASE_NOT_CONFIGURED');
  const [campaign] = await db.select().from(campaigns)
    .where(and(eq(campaigns.id, campaignId), eq(campaigns.tenantId, tenantId))).limit(1);
  if (!campaign) throw new Error('CAMPAIGN_NOT_FOUND');
  if (campaign.status === 'sent') throw new Error('ALREADY_SENT');

  const [tenant] = await db.select().from(tenants).where(eq(tenants.id, tenantId)).limit(1);
  const audience = await buildAudience(tenantId, campaign.type as CampaignType);

  await db.update(campaigns).set({ status: 'sending', audienceCount: audience.length }).where(eq(campaigns.id, campaignId));

  let sent = 0, failed = 0;
  for (const guest of audience) {
    const msg = personalize(campaign.messageTemplate, guest, tenant);
    try {
      await sendMessageToGuest(tenantId, guest.phone, msg, { guestId: guest.id, campaignId, handledBy: 'campaign' });
      sent++;
      // Record recipient so later redemptions can be attributed
      try {
        await db.insert(campaignRecipients).values({ campaignId, guestId: guest.id, tenantId });
      } catch (e) { console.warn('[campaign] recipient log failed', e); }
    } catch (e) { failed++; console.error('[campaign] send failed to', guest.phone, e); }
  }

  await db.update(campaigns).set({ status: 'sent', sentAt: new Date() }).where(eq(campaigns.id, campaignId));
  return { sent, failed, total: audience.length };
}
```

```ts
export const ATTRIBUTION_WINDOW_DAYS = 7;

/** Last-touch attribution: credit the most recent campaign sent to this guest within the window. */
export async function attributeRedemptionToCampaign(tenantId: string, guestId: string): Promise<string | null> {
  if (!db) return null;
  const windowStart = new Date(Date.now() - ATTRIBUTION_WINDOW_DAYS * 24 * 3600 * 1000);

  const [recipient] = await db.select().from(campaignRecipients)
    .where(and(
      eq(campaignRecipients.tenantId, tenantId),
      eq(campaignRecipients.guestId, guestId),
      gte(campaignRecipients.sentAt, windowStart),
    ))
    .orderBy(desc(campaignRecipients.sentAt))
    .limit(1);
  if (!recipient) return null;

  const [tenant] = await db.select().from(tenants).where(eq(tenants.id, tenantId)).limit(1);
  const revenuePerRedemption = tenant?.avgSpend ?? 25000; // a redeemed reward = a filled seat ≈ avg spend

  await db.update(campaigns)
    .set({
      redemptionsCount: sql`${campaigns.redemptionsCount} + 1`,
      revenueGenerated: sql`${campaigns.revenueGenerated} + ${revenuePerRedemption}`,
    })
    .where(eq(campaigns.id, recipient.campaignId));

  return recipient.campaignId;
}
```

```ts
export interface EvolutionLifecycleResult { ok: boolean; data?: any; error?: string; }

export async function createInstance(instanceName: string): Promise<EvolutionLifecycleResult> {
  const base = apiUrl(); const key = globalKey();
  if (!base || !key) return { ok: false, error: 'EVOLUTION_NOT_CONFIGURED' };
  try {
    const res = await fetch(`${base}/instance/create`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', apikey: key },
      body: JSON.stringify({ instanceName, qrcode: true, integration: 'WHATSAPP-BAILEYS' }),
    });
    const data = await res.json().catch(() => ({}));
    if (!res.ok) return { ok: false, error: data?.message || `CREATE_FAILED_${res.status}`, data };
    return { ok: true, data };
  } catch (e: any) { return { ok: false, error: e?.message || 'CREATE_ERROR' }; }
}

export async function getQrCode(instanceName: string): Promise<EvolutionLifecycleResult> {
  const base = apiUrl(); const key = globalKey();
  if (!base || !key) return { ok: false, error: 'EVOLUTION_NOT_CONFIGURED' };
  try {
    const res = await fetch(`${base}/instance/connect/${instanceName}`, { headers: { apikey: key } });
    const data = await res.json().catch(() => ({}));
    if (!res.ok) return { ok: false, error: data?.message || `CONNECT_FAILED_${res.status}`, data };
    return { ok: true, data };
  } catch (e: any) { return { ok: false, error: e?.message || 'CONNECT_ERROR' }; }
}
```

```ts
// src/app/api/v1/whatsapp/connect/route.ts
import { NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { tenants } from '@/lib/db/schema';
import { eq } from 'drizzle-orm';
import { requireTenantContext } from '@/shared/utils/tenant-context';
import { createInstance, getQrCode, extractQrBase64, extractApiKey } from '@/lib/integrations/evolution/lifecycle';

function makeInstanceName(tenantId: string): string {
  return `t_${tenantId.replace(/-/g, '').slice(0, 16)}`;
}

export async function POST() {
  try {
    const { tenant } = await requireTenantContext();
    if (!db) return NextResponse.json({ error: { code: 'DATABASE_NOT_CONFIGURED' } }, { status: 503 });

    let qr: string | null = null;

    if (tenant.whatsappInstanceName) {
      // Re-connect an existing instance
      const conn = await getQrCode(tenant.whatsappInstanceName);
      if (!conn.ok) return NextResponse.json({ error: { code: 'CONNECT_FAILED', detail: conn.error } }, { status: 502 });
      qr = extractQrBase64(conn.data);
    } else {
      // Create a new per-tenant instance; store the per-tenant token for sending
      const instanceName = makeInstanceName(tenant.id);
      const created = await createInstance(instanceName);
      if (!created.ok) return NextResponse.json({ error: { code: 'CREATE_FAILED', detail: created.error } }, { status: 502 });
      const token = extractApiKey(created.data);
      qr = extractQrBase64(created.data);
      await db.update(tenants).set({
        whatsappInstanceName: instanceName,
        whatsappInstanceToken: token,
        whatsappConnected: false,
      }).where(eq(tenants.id, tenant.id));
    }

    if (!qr) return NextResponse.json({ error: { code: 'NO_QR' } }, { status: 502 });
    return NextResponse.json({ ok: true, qr });
  } catch (e) {
    const msg = e instanceof Error ? e.message : 'INTERNAL';
    const status = msg === 'UNAUTHORIZED' ? 401 : msg === 'NO_TENANT' ? 403 : 500;
    return NextResponse.json({ error: { code: msg } }, { status });
  }
}
```

## 2 · The selftest route

```ts
// src/app/api/v1/selftest/route.ts
import { NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { tenants, guests, loyaltyTransactions, rewardEvents, rewardsCatalog } from '@/lib/db/schema';
import { sql } from 'drizzle-orm';
import { CLAIM_RADIUS_METERS, haversineMeters } from '@/modules/rewards/service';
import { EARN_CENTS_PER_POINT, FLAT_VISIT_POINTS } from '@/modules/loyalty/service';
import { CAMPAIGN_PRESETS, buildAudience, estimateRoi } from '@/modules/campaigns/service';

export const dynamic = 'force-dynamic';

type Status = 'pass' | 'warn' | 'fail' | 'skip';
interface Check { step: string; status: Status; detail?: any }

async function run(step: string, fn: () => Promise<{ status: Status; detail?: any }>): Promise<Check> {
  try {
    const r = await fn();
    return { step, ...r };
  } catch (e: any) {
    return { step, status: 'fail', detail: e?.message || 'unknown error' };
  }
}

export async function GET(request: Request) {
  const secret = process.env.SELFTEST_SECRET || process.env.CRON_SECRET;
  const auth = request.headers.get('authorization') ?? '';
  if (!secret || (auth !== `Bearer ${secret}` && auth !== secret)) {
    return NextResponse.json({ error: { code: 'UNAUTHORIZED' } }, { status: 401 });
  }

  const checks: Check[] = [];

  // 1) Config
  checks.push(await run('config', async () => {
    const required = ['DATABASE_URL', 'NEXT_PUBLIC_APP_URL', 'EVOLUTION_API_URL', 'EVOLUTION_GLOBAL_API_KEY'];
    const missing = required.filter((k) => !process.env[k]);
    const optionalMissing = ['OPENAI_API_KEY'].filter((k) => !process.env[k]);
    if (missing.length) return { status: 'fail', detail: { missing } };
    return { status: optionalMissing.length ? 'warn' : 'pass', detail: { requiredPresent: required.length, optionalMissing } };
  }));

  // 2) Auth (Clerk)
  checks.push(await run('auth', async () => {
    const pk = process.env.NEXT_PUBLIC_CLERK_PUBLISHABLE_KEY;
    const sk = process.env.CLERK_SECRET_KEY;
    if (!pk || !sk) return { status: 'fail', detail: 'Clerk keys missing' };
    return { status: 'pass', detail: 'Clerk configured (publishable + secret present)' };
  }));

  // 3) Database
  checks.push(await run('database', async () => {
    if (!db) return { status: 'fail', detail: 'db client is null (DATABASE_URL not set at runtime)' };
    const [row] = await db.select({ count: sql<number>`count(*)::int` }).from(tenants);
    return { status: 'pass', detail: { tenants: row?.count ?? 0 } };
  }));

  // 4) WhatsApp (Evolution) — tolerant of free-tier sleep
  checks.push(await run('whatsapp', async () => {
    const url = process.env.EVOLUTION_API_URL;
    const key = process.env.EVOLUTION_GLOBAL_API_KEY;
    if (!url || !key) return { status: 'fail', detail: 'Evolution config missing' };
    try {
      const ctrl = new AbortController();
      const t = setTimeout(() => ctrl.abort(), 5000);
      const res = await fetch(url, { signal: ctrl.signal, headers: { apikey: key } });
      clearTimeout(t);
      return { status: 'pass', detail: { reachable: true, http: res.status } };
    } catch {
      return { status: 'warn', detail: { reachable: false, note: 'Evolution may be sleeping (free tier). Config is present.' } };
    }
  }));

  // 5) Loyalty — earn calc + table reads
  checks.push(await run('loyalty', async () => {
    const sampleSpendCents = 45000; // R450
    const earned = Math.max(1, Math.floor(sampleSpendCents / EARN_CENTS_PER_POINT));
    if (earned <= 0) return { status: 'fail', detail: 'earn calculation produced 0' };
    let guestCount = null, txCount = null;
    if (db) {
      const [g] = await db.select({ count: sql<number>`count(*)::int` }).from(guests);
      const [lt] = await db.select({ count: sql<number>`count(*)::int` }).from(loyaltyTransactions);
      guestCount = g?.count ?? 0; txCount = lt?.count ?? 0;
    }
    return { status: 'pass', detail: { earnCalc: { spendCents: sampleSpendCents, earned, centsPerPoint: EARN_CENTS_PER_POINT, flatVisit: FLAT_VISIT_POINTS }, guests: guestCount, transactions: txCount } };
  }));

  // 6) Campaigns — presets + read-only audience/ROI
  checks.push(await run('campaign', async () => {
    const expected = ['fill_quiet_hours', 'bring_back_lost', 'reward_vips'];
    const missingPresets = expected.filter((t) => !CAMPAIGN_PRESETS[t as keyof typeof CAMPAIGN_PRESETS]);
    if (missingPresets.length) return { status: 'fail', detail: { missingPresets } };
    for (const t of expected) {
      const p = CAMPAIGN_PRESETS[t as keyof typeof CAMPAIGN_PRESETS];
      if (!p.statuses?.length || !p.defaultMessage || !p.conversionRate) return { status: 'fail', detail: `preset ${t} incomplete` };
    }
    if (db) {
      const [t] = await db.select({ id: tenants.id }).from(tenants).limit(1);
      if (t) {
        const audience = await buildAudience(t.id, 'reward_vips');
        const roi = await estimateRoi(t.id, 'reward_vips', audience.length);
        return { status: 'pass', detail: { presets: expected.length, sampleAudience: audience.length, sampleRoiCents: roi.estimatedRevenueCents } };
      }
    }
    return { status: 'pass', detail: { presets: expected.length, note: 'no tenant yet — audience/ROI skipped' } };
  }));

  // 7) Claim — haversine + radius + table reads
  checks.push(await run('claim', async () => {
    // ~500m apart (same lon, lat differs by 0.0045°)
    const d = haversineMeters(-33.9249, 18.4241, -33.9294, 18.4241);
    if (!(d > 300 && d < 700)) return { status: 'fail', detail: { distanceM: Math.round(d), note: 'haversine out of expected range' } };
    if (CLAIM_RADIUS_METERS !== 500) return { status: 'warn', detail: { radiusM: CLAIM_RADIUS_METERS, note: 'claim radius is not 500m' } };
    let eventCount = null, rewardCount = null;
    if (db) {
      const [re] = await db.select({ count: sql<number>`count(*)::int` }).from(rewardEvents);
      const [rc] = await db.select({ count: sql<number>`count(*)::int` }).from(rewardsCatalog);
      eventCount = re?.count ?? 0; rewardCount = rc?.count ?? 0;
    }
    return { status: 'pass', detail: { haversineSampleM: Math.round(d), radiusM: CLAIM_RADIUS_METERS, rewardEvents: eventCount, rewards: rewardCount } };
  }));

  const failed = checks.filter((c) => c.status === 'fail').map((c) => c.step);
  const healthy = failed.length === 0;
  return NextResponse.json({
    healthy,
    failed,
    passed: checks.filter((c) => c.status === 'pass').length,
    warned: checks.filter((c) => c.status === 'warn').length,
    checks,
    timestamp: new Date().toISOString(),
  });
}
```
